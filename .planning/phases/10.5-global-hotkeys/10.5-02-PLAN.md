```csharp
using System.Runtime.InteropServices;
using MatrixShader.Core.Native;

namespace MatrixShader.Hotkeys;

/// <summary>
/// Hidden message-only window for receiving global hotkey messages.
/// Uses pure P/Invoke - no WinForms dependency for AOT compatibility.
/// </summary>
public sealed class HotkeyWindow : IDisposable
{
    private nint _hwnd;
    private bool _disposed;
    private readonly Action<int> _hotkeyHandler;
    private WndProcDelegate? _wndProc; // Must keep reference to prevent GC

    // Delegate for window procedure
    private delegate nint WndProcDelegate(nint hWnd, uint msg, nint wParam, nint lParam);

    public nint Handle => _hwnd;

    public HotkeyWindow(Action<int> hotkeyHandler)
    {
        _hotkeyHandler = hotkeyHandler ?? throw new ArgumentNullException(nameof(hotkeyHandler));
    }

    /// <summary>
    /// Creates the hidden message-only window.
    /// Must be called before RegisterHotKey and RunMessageLoop.
    /// </summary>
    public bool Create()
    {
        // Keep delegate reference to prevent GC collection
        _wndProc = WndProc;

        // Register window class
        var className = $"MatrixHotkeyWindow_{Environment.ProcessId}";
        var wndClass = new WNDCLASSEXW
        {
            cbSize = (uint)Marshal.SizeOf<WNDCLASSEXW>(),
            lpfnWndProc = Marshal.GetFunctionPointerForDelegate(_wndProc),
            hInstance = HotkeyApi.GetModuleHandleW(null),
            lpszClassName = className
        };

        var atom = HotkeyApi.RegisterClassExW(ref wndClass);
        if (atom == 0)
            return false;

        // Create message-only window (HWND_MESSAGE parent)
        _hwnd = HotkeyApi.CreateWindowExW(
            0,                      // dwExStyle
            className,              // lpClassName
            "MatrixHotkeys",        // lpWindowName
            0,                      // dwStyle
            0, 0, 0, 0,            // position/size (ignored for message-only)
            HotkeyApi.HWND_MESSAGE, // hWndParent - message-only window
            nint.Zero,             // hMenu
            wndClass.hInstance,    // hInstance
            nint.Zero              // lpParam
        );

        return _hwnd != nint.Zero;
    }

    /// <summary>
    /// Runs the message loop. Blocks until Stop() is called or window destroyed.
    /// MUST be called on the same thread that called Create().
    /// </summary>
    public void RunMessageLoop()
    {
        while (HotkeyApi.GetMessageW(out var msg, nint.Zero, 0, 0))
        {
            HotkeyApi.TranslateMessage(ref msg);
            HotkeyApi.DispatchMessageW(ref msg);
        }
    }

    /// <summary>
    /// Posts quit message to stop the message loop.
    /// Thread-safe - can be called from any thread.
    /// </summary>
    public void Stop()
    {
        if (_hwnd != nint.Zero)
        {
            HotkeyApi.PostMessageW(_hwnd, HotkeyApi.WM_QUIT, nint.Zero, nint.Zero);
        }
    }

    private nint WndProc(nint hWnd, uint msg, nint wParam, nint lParam)
    {
        if (msg == HotkeyApi.WM_HOTKEY)
        {
            var hotkeyId = (int)wParam;
            _hotkeyHandler(hotkeyId);
            return nint.Zero;
        }

        return HotkeyApi.DefWindowProcW(hWnd, msg, wParam, lParam);
    }

    public void Dispose()
    {
        if (_disposed) return;
        _disposed = true;

        if (_hwnd != nint.Zero)
        {
            HotkeyApi.DestroyWindow(_hwnd);
            _hwnd = nint.Zero;
        }
    }
}
```

**HotkeyManager.cs** - Manages hotkey registration lifecycle:

```csharp
using MatrixShader.Core.Models;
using MatrixShader.Core.Native;

namespace MatrixShader.Hotkeys;

/// <summary>
/// Manages hotkey registration/unregistration with conflict tracking.
/// </summary>
public sealed class HotkeyManager : IDisposable
{
    private readonly nint _windowHandle;
    private readonly Dictionary<int, (HotkeyAction Action, Action Handler)> _registeredHotkeys = new();
    private readonly List<HotkeyBinding> _failedHotkeys = new();
    private int _nextId = 1;
    private bool _disposed;

    public HotkeyManager(nint windowHandle)
    {
        _windowHandle = windowHandle;
    }

    /// <summary>
    /// Registers a hotkey. Returns false if registration fails (already taken).
    /// </summary>
    public bool RegisterHotkey(HotkeyBinding binding, Action handler)
    {
        if (!binding.Enabled)
            return true; // Disabled hotkeys succeed silently

        // Always include MOD_NOREPEAT to prevent auto-repeat spam
        var modifiers = binding.Modifiers | HotkeyApi.MOD_NOREPEAT;

        var id = _nextId++;
        if (HotkeyApi.RegisterHotKey(_windowHandle, id, modifiers, binding.VirtualKey))
        {
            _registeredHotkeys[id] = (binding.Action, handler);
            return true;
        }

        _failedHotkeys.Add(binding);
        return false;
    }

    /// <summary>
    /// Unregisters all hotkeys. Called on dispose.
    /// </summary>
    public void UnregisterAll()
    {
        foreach (var id in _registeredHotkeys.Keys)
        {
            HotkeyApi.UnregisterHotKey(_windowHandle, id);
        }
        _registeredHotkeys.Clear();
    }

    /// <summary>
    /// Returns list of hotkeys that failed to register (conflicts).
    /// </summary>
    public IReadOnlyList<HotkeyBinding> GetFailedHotkeys() => _failedHotkeys;

    /// <summary>
    /// Called by HotkeyWindow when WM_HOTKEY received.
    /// </summary>
    public void HandleHotkey(int id)
    {
        if (_registeredHotkeys.TryGetValue(id, out var entry))
        {
            try
            {
                entry.Handler();
            }
            catch
            {
                // Silent failure per CONTEXT.md - "if action can't complete, do nothing"
            }
        }
    }

    public void Dispose()
    {
        if (_disposed) return;
        _disposed = true;
        UnregisterAll();
    }
}
```

namespace MatrixShader.Hotkeys;

/// <summary>
/// Enforces single-instance using named mutex.
/// </summary>
public static class SingleInstance
{
    // Global prefix ensures cross-session detection
    // GUID ensures no collision with other apps
    private const string MutexName = "Global\\MatrixShader.Hotkeys.{8A9B1C2D-3E4F-5A6B-7C8D-9E0F1A2B3C4D}";
    private static Mutex? _mutex;

    /// <summary>
    /// Attempts to acquire single-instance lock.
    /// Returns true if this is the only instance, false if another exists.
    /// </summary>
    public static bool TryAcquire()
    {
        _mutex = new Mutex(true, MutexName, out bool createdNew);
        if (!createdNew)
        {
            _mutex.Dispose();
            _mutex = null;
            return false; // Another instance is running
        }
        return true;
    }

    /// <summary>
    /// Releases the single-instance lock.
    /// </summary>
    public static void Release()
    {
        if (_mutex != null)
        {
            try
            {
                _mutex.ReleaseMutex();
            }
            catch (ApplicationException)
            {
                // Mutex wasn't owned - ignore
            }
            _mutex.Dispose();
            _mutex = null;
        }
    }
}
