```csharp
    // Shader configuration - NOTE: JSON property name is "experimental.pixelShaderPath"
    // Use JsonPropertyName attribute for the dotted property name
    [JsonPropertyName("experimental.pixelShaderPath")]
    public string? PixelShaderPath { get; init; }

    // Tab color (hex format like "#00FF4C")
    public string? TabColor { get; init; }
}
```

```csharp
using MatrixShader.Core.Models;
using MatrixShader.Core.Native;
using MatrixShader.Core.Services;

namespace MatrixShader.Hotkeys;

/// <summary>
/// Implements all hotkey action handlers.
/// Uses existing Core services for actual functionality.
/// </summary>
public class HotkeyActions
{
    private readonly ILayoutService _layoutService;
    private readonly IShaderService _shaderService;
    private readonly IConfigService _configService;
    private readonly IIdentityService _identityService;
    private readonly ITerminalSettingsService _terminalSettingsService;

    // Available shader files (Matrix-1 through Matrix-8)
    private static readonly string[] ShaderFiles = Enumerable.Range(1, 8)
        .Select(i => $"Matrix-{i}.hlsl")
        .ToArray();

    public HotkeyActions(
        ILayoutService layoutService,
        IShaderService shaderService,
        IConfigService configService,
        IIdentityService identityService,
        ITerminalSettingsService terminalSettingsService)
    {
        _layoutService = layoutService;
        _shaderService = shaderService;
        _configService = configService;
        _identityService = identityService;
        _terminalSettingsService = terminalSettingsService;
    }

    /// <summary>
    /// Gets the handler for a specific hotkey action.
    /// </summary>
    public Action GetHandler(HotkeyAction action) => action switch
    {
        HotkeyAction.SwapLeft => SwapLeft,
        HotkeyAction.SwapRight => SwapRight,
        HotkeyAction.CycleLayout => CycleLayout,
        HotkeyAction.ToggleTransparency => ToggleTransparency,
        HotkeyAction.OpacityDown => OpacityDown,
        HotkeyAction.OpacityUp => OpacityUp,
        HotkeyAction.CycleShader => CycleShader,
        HotkeyAction.SpeedUp => SpeedUp,
        HotkeyAction.SpeedDown => SpeedDown,
        HotkeyAction.ToggleFar => ToggleFar,
        HotkeyAction.ToggleMid => ToggleMid,
        HotkeyAction.ToggleNear => ToggleNear,
        _ => () => { } // No-op for unknown actions
    };

    // ===== Window Actions =====

    private void SwapLeft()
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        var windows = _identityService.FindMatrixWindows();
        var sorted = windows.OrderBy(w => w.ShaderIndex).ToList();
        var index = sorted.FindIndex(w => w.Handle == focusedWindow.Handle);

        if (index > 0)
        {
            SwapWindowPositions(sorted[index], sorted[index - 1]);
        }
    }

    private void SwapRight()
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        var windows = _identityService.FindMatrixWindows();
        var sorted = windows.OrderBy(w => w.ShaderIndex).ToList();
        var index = sorted.FindIndex(w => w.Handle == focusedWindow.Handle);

        if (index >= 0 && index < sorted.Count - 1)
        {
            SwapWindowPositions(sorted[index], sorted[index + 1]);
        }
    }

    private void CycleLayout()
    {
        var state = _configService.LoadState();
        var currentMode = state.Layout.Mode;

        // Cycle: Pillars -> Quads -> Auto -> Pillars
        var nextMode = currentMode switch
        {
            "Pillars" => "Quads",
            "Quads" => "Auto",
            _ => "Pillars"
        };

        var newState = state with
        {
            Layout = state.Layout with { Mode = nextMode }
        };
        _configService.SaveState(newState);

        // Apply layout immediately
        var windows = _identityService.FindMatrixWindows();
        var positions = _layoutService.CalculateLayout(windows, newState.Layout);
        _layoutService.ApplyLayout(positions);
    }

    private void ToggleTransparency()
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        try
        {
            var settings = _terminalSettingsService.LoadSettings();
            var profileName = $"Matrix-{focusedWindow.ShaderIndex}";
            var profile = _terminalSettingsService.GetProfile(settings, profileName);

            if (profile == null) return;

            // Toggle useAcrylic
            var updatedProfile = profile with { UseAcrylic = !profile.UseAcrylic };
            _terminalSettingsService.UpsertProfile(settings, updatedProfile);
            _terminalSettingsService.SaveSettings(settings);
        }
        catch
        {
            // Silent failure per CONTEXT.md
        }
    }

    // ===== Opacity Actions =====

    private void OpacityDown()
    {
        AdjustOpacity(-5);
    }

    private void OpacityUp()
    {
        AdjustOpacity(5);
    }

    private void AdjustOpacity(int delta)
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        try
        {
            var settings = _terminalSettingsService.LoadSettings();
            var profileName = $"Matrix-{focusedWindow.ShaderIndex}";
            var profile = _terminalSettingsService.GetProfile(settings, profileName);

            if (profile == null) return;

            // Clamp opacity between 0 and 100
            var newOpacity = Math.Clamp(profile.Opacity + delta, 0, 100);
            var updatedProfile = profile with { Opacity = newOpacity };

            _terminalSettingsService.UpsertProfile(settings, updatedProfile);
            _terminalSettingsService.SaveSettings(settings);
        }
        catch
        {
            // Silent failure per CONTEXT.md
        }
    }

    // ===== Shader Actions =====

    private void CycleShader()
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        try
        {
            var settings = _terminalSettingsService.LoadSettings();
            var profileName = $"Matrix-{focusedWindow.ShaderIndex}";
            var profile = _terminalSettingsService.GetProfile(settings, profileName);

            if (profile == null || string.IsNullOrEmpty(profile.PixelShaderPath)) return;

            // Get current shader filename and find next in cycle
            var currentFilename = Path.GetFileName(profile.PixelShaderPath);
            var currentIndex = Array.FindIndex(ShaderFiles, f =>
                string.Equals(f, currentFilename, StringComparison.OrdinalIgnoreCase));

            // Cycle to next shader (wrap around)
            var nextIndex = (currentIndex + 1) % ShaderFiles.Length;
            var nextFilename = ShaderFiles[nextIndex];

            // Update path keeping same directory
            var shaderDir = Path.GetDirectoryName(profile.PixelShaderPath) ?? "";
            var newPath = Path.Combine(shaderDir, nextFilename);

            var updatedProfile = profile with { PixelShaderPath = newPath };
            _terminalSettingsService.UpsertProfile(settings, updatedProfile);
            _terminalSettingsService.SaveSettings(settings);
        }
        catch
        {
            // Silent failure per CONTEXT.md
        }
    }

    private void SpeedUp()
    {
        AdjustSpeed(0.1f);
    }

    private void SpeedDown()
    {
        AdjustSpeed(-0.1f);
    }

    private void AdjustSpeed(float delta)
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        var state = _configService.LoadState();
        if (!state.ShaderConfigs.TryGetValue(focusedWindow.ShaderIndex, out var config))
            return;

        var newSpeed = Math.Clamp(config.RainSpeed + delta, 0.1f, 5.0f);
        var newConfig = config with { RainSpeed = newSpeed };

        var newConfigs = new Dictionary<int, ShaderConfig>(state.ShaderConfigs)
        {
            [focusedWindow.ShaderIndex] = newConfig
        };

        _configService.SaveState(state with { ShaderConfigs = newConfigs });
        _shaderService.WriteShader(focusedWindow.ShaderIndex, newConfig);
    }

    // ===== Layer Actions =====

    private void ToggleFar()
    {
        ToggleLayer(c => c with { ShowL1 = !c.ShowL1 });
    }

    private void ToggleMid()
    {
        ToggleLayer(c => c with { ShowL2 = !c.ShowL2 });
    }

    private void ToggleNear()
    {
        ToggleLayer(c => c with { ShowL3 = !c.ShowL3 });
    }

    private void ToggleLayer(Func<ShaderConfig, ShaderConfig> toggle)
    {
        var focusedWindow = GetFocusedMatrixWindow();
        if (focusedWindow == null) return;

        var state = _configService.LoadState();
        if (!state.ShaderConfigs.TryGetValue(focusedWindow.ShaderIndex, out var config))
            return;

        var newConfig = toggle(config);
        var newConfigs = new Dictionary<int, ShaderConfig>(state.ShaderConfigs)
        {
            [focusedWindow.ShaderIndex] = newConfig
        };

        _configService.SaveState(state with { ShaderConfigs = newConfigs });
        _shaderService.WriteShader(focusedWindow.ShaderIndex, newConfig);
    }

    // ===== Helpers =====

    private WindowInfo? GetFocusedMatrixWindow()
    {
        var foreground = WindowsApi.GetForegroundWindow();
        if (foreground == nint.Zero) return null;

        return _identityService.ResolveIdentity(foreground);
    }

    private void SwapWindowPositions(WindowInfo window1, WindowInfo window2)
    {
        // Get current positions
        if (!WindowsApi.GetWindowRect(window1.Handle, out var rect1)) return;
        if (!WindowsApi.GetWindowRect(window2.Handle, out var rect2)) return;

        // Swap positions
        WindowsApi.SetWindowPos(window1.Handle, nint.Zero,
            rect2.Left, rect2.Top,
            rect2.Right - rect2.Left, rect2.Bottom - rect2.Top,
            WindowsApi.SWP_NOZORDER | WindowsApi.SWP_NOACTIVATE);

        WindowsApi.SetWindowPos(window2.Handle, nint.Zero,
            rect1.Left, rect1.Top,
            rect1.Right - rect1.Left, rect1.Bottom - rect1.Top,
            WindowsApi.SWP_NOZORDER | WindowsApi.SWP_NOACTIVATE);
    }
}
```

```csharp
using MatrixShader.Core.Models;
using Microsoft.Toolkit.Uwp.Notifications;

namespace MatrixShader.Hotkeys;

/// <summary>
/// Toast notifications for hotkey conflicts and errors.
/// Uses Microsoft.Toolkit.Uwp.Notifications for unpackaged app support.
/// </summary>
public static class ToastNotifications
{
    /// <summary>
    /// Shows toast notification listing failed hotkey registrations.
    /// Per CONTEXT.md: "Toast directs user to configure alternatives in Redpill"
    /// </summary>
    public static void ShowConflictWarning(IReadOnlyList<HotkeyBinding> failedHotkeys)
    {
        if (failedHotkeys.Count == 0) return;

        var hotkeyList = string.Join(", ", failedHotkeys.Select(h => h.DisplayName));

        try
        {
            new ToastContentBuilder()
                .AddText("Matrix Hotkeys: Some shortcuts unavailable")
                .AddText($"Failed to register: {hotkeyList}")
                .AddText("Run 'redpill --hotkeys' to configure alternatives")
                .Show();
        }
        catch
        {
            // Silent failure - toast not critical
            // May fail on systems without toast support
        }
    }

    /// <summary>
    /// Cleanup for non-MSIX apps. Call on process exit.
    /// </summary>
    public static void Cleanup()
    {
        try
        {
            // Uninstall removes the app's AUMID from the registry
            // Required for non-packaged apps per documentation
            ToastNotificationManagerCompat.Uninstall();
        }
        catch
        {
            // Cleanup is best-effort
        }
    }
}
```

IMPORTANT: The toast notification package requires the TFM to be net8.0-windows10.0.17763.0 or later, which was set in Plan 02. If AOT trimming causes issues, wrap in try/catch and degrade gracefully (no toast).
