```pascal
procedure CurUninstallStepChanged(CurUninstallStep: TUninstallStep);
var
  AppDir: string;
  LocalAppDataDir: string;
  ErrorMsg: string;
begin
  if CurUninstallStep = usPostUninstall then
  begin
    // Get directories
    AppDir := ExpandConstant('{app}');
    LocalAppDataDir := ExpandConstant('{localappdata}\MatrixShader');

    // Remove Matrix profiles from Windows Terminal settings.json (best effort)
    // This is done separately in case WT is not installed

    // Verify cleanup
    if DirExists(AppDir) then
    begin
      ErrorMsg := 'Some files could not be removed from:' + #13#10 +
                  AppDir + #13#10 + #13#10 +
                  'This may be because:' + #13#10 +
                  '- Files are in use (close all Matrix windows first)' + #13#10 +
                  '- Permission denied (try running as Administrator)' + #13#10 + #13#10 +
                  'To remove manually:' + #13#10 +
                  '1. Close all Matrix Shader windows' + #13#10 +
                  '2. Delete the folder: ' + AppDir;
      MsgBox(ErrorMsg, mbError, MB_OK);
    end;

    if DirExists(LocalAppDataDir) then
    begin
      // Less critical - just warn
      MsgBox('User data folder could not be fully removed:' + #13#10 +
             LocalAppDataDir + #13#10 + #13#10 +
             'You can delete this folder manually if desired.',
             mbInformation, MB_OK);
    end;
  end;
end;
```

Note: Inno Setup's `Type: filesandordirs` should handle recursive deletion. If files are in use, the explicit error message helps users understand what to do.

```pascal
function InitializeSetup(): Boolean;
var
  ExistingDir: string;
  ResultCode: Integer;
  Choice: Integer;
begin
  Result := True;

  // Check if already installed
  ExistingDir := ExpandConstant('{autopf}\MatrixShader');
  if DirExists(ExistingDir) then
  begin
    // Check if any executables exist (confirm it's a real install)
    if FileExists(ExistingDir + '\wakeupneo.exe') or
       FileExists(ExistingDir + '\bluepill.exe') or
       FileExists(ExistingDir + '\redpill.exe') then
    begin
      Choice := MsgBox('Matrix Shader is already installed.' + #13#10 + #13#10 +
                       'What would you like to do?' + #13#10 + #13#10 +
                       '  [Yes] = Update/Repair (recommended)' + #13#10 +
                       '  [No] = Uninstall first, then reinstall' + #13#10 +
                       '  [Cancel] = Exit installer',
                       mbConfirmation, MB_YESNOCANCEL);

      case Choice of
        IDYES:
          begin
            // Continue with update/repair (Inno Setup handles this)
            Result := True;
          end;
        IDNO:
          begin
            // Run uninstaller first
            if FileExists(ExistingDir + '\unins000.exe') then
            begin
              MsgBox('The uninstaller will now run.' + #13#10 +
                     'After uninstall completes, run this installer again.',
                     mbInformation, MB_OK);
              Exec(ExistingDir + '\unins000.exe', '', '', SW_SHOW, ewWaitUntilTerminated, ResultCode);
            end
            else
            begin
              MsgBox('Uninstaller not found. Please uninstall via Control Panel first.',
                     mbError, MB_OK);
            end;
            Result := False; // Exit this installer
          end;
        IDCANCEL:
          begin
            Result := False; // Exit installer
          end;
      end;
    end;
  end;
end;
```

This uses Inno Setup's built-in message box with Yes/No/Cancel which maps well to Update/Uninstall/Cancel.
