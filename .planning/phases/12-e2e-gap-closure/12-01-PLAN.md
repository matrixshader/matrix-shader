  <name>Task 1: Fix shader regex replacement bug (BUG-SHADER01)</name>
  <files>MatrixShader/src/MatrixShader.Core/Services/ShaderService.cs</files>
  <action>
Fix the ReplaceDefine method at line ~204. The bug is that `$1` in the replacement string is interpreted as a regex backreference.

Current broken code:
```csharp
return Regex.Replace(content, pattern, $"$1{value.ToString("F1", CultureInfo.InvariantCulture)}");
```

The issue: When value=0.0, the replacement becomes `$10.0` which regex interprets as "capture group 10" (doesn't exist) + ".0".

Fix: Use a MatchEvaluator delegate or escape the replacement:

```csharp
private static string ReplaceDefine(string content, string name, float value)
{
    var pattern = $@"(#define\s+{name}\s+)[\d.]+";
    var replacement = value.ToString("F1", CultureInfo.InvariantCulture);
    return Regex.Replace(content, pattern, m => m.Groups[1].Value + replacement);
}
```

Using a MatchEvaluator avoids all replacement string interpretation issues.

```csharp
/// <summary>
/// Windows Terminal installation locations to check.
/// </summary>
private static readonly string[] WtSettingsPaths = new[]
{
    // Microsoft Store install
    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
        "Packages", "Microsoft.WindowsTerminal_8wekyb3d8bbwe", "LocalState", "settings.json"),
    // Winget / unpackaged install
    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
        "Microsoft", "Windows Terminal", "settings.json"),
    // Scoop install
    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile),
        "scoop", "persist", "windows-terminal", "settings", "settings.json"),
    // Chocolatey install
    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.CommonApplicationData),
        "chocolatey", "lib", "microsoft-windows-terminal", "settings.json"),
};

/// <summary>
/// Checks if Windows Terminal is installed by checking multiple known paths.
/// </summary>
public static bool IsWindowsTerminalInstalled()
{
    // Check settings files
    if (WtSettingsPaths.Any(File.Exists))
        return true;

    // Check if wt.exe is in PATH
    if (GetWindowsTerminalExePath() != null)
        return true;

    // Check parent process (running inside WT)
    if (EnvironmentService.IsWindowsTerminal())
        return true;

    return false;
}

/// <summary>
/// Gets the path to settings.json for the detected WT install.
/// </summary>
public static string? GetSettingsPath()
{
    return WtSettingsPaths.FirstOrDefault(File.Exists);
}

/// <summary>
/// Finds wt.exe path dynamically.
/// </summary>
public static string? GetWindowsTerminalExePath()
{
    // Check PATH first
    var pathDirs = Environment.GetEnvironmentVariable("PATH")?.Split(Path.PathSeparator) ?? Array.Empty<string>();
    foreach (var dir in pathDirs)
    {
        var wtPath = Path.Combine(dir, "wt.exe");
        if (File.Exists(wtPath))
            return wtPath;
    }

    // Check common install locations
    var commonPaths = new[]
    {
        // Store app (WindowsApps - may not be directly accessible but worth checking)
        Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ProgramFiles),
            "WindowsApps", "Microsoft.WindowsTerminal_*", "wt.exe"),
        // Scoop
        Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile),
            "scoop", "apps", "windows-terminal", "current", "wt.exe"),
        // Chocolatey
        Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.CommonApplicationData),
            "chocolatey", "bin", "wt.exe"),
    };

    foreach (var pattern in commonPaths)
    {
        if (pattern.Contains("*"))
        {
            // Handle wildcard paths
            var dir = Path.GetDirectoryName(pattern);
            var file = Path.GetFileName(pattern);
            if (dir != null && Directory.Exists(Path.GetDirectoryName(dir)))
            {
                try
                {
                    var matches = Directory.GetDirectories(Path.GetDirectoryName(dir)!, Path.GetFileName(dir));
                    foreach (var match in matches)
                    {
                        var wtPath = Path.Combine(match, file);
                        if (File.Exists(wtPath))
                            return wtPath;
                    }
                }
                catch { /* Access denied - expected for WindowsApps */ }
            }
        }
        else if (File.Exists(pattern))
        {
            return pattern;
        }
    }

    // Fallback: assume wt.exe is in PATH (will fail at launch time if not)
    return null;
}
```
