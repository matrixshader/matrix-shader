<task type="auto">
  <name>Task 2: Create validation script</name>
  <files>installer/test/validate.ps1</files>
  <action>
    Create validate.ps1 that tests all requirements:

    1. Verify each executable exists
    2. Measure startup time (run with --help to avoid interactive mode)
    3. Verify startup is under 500ms
    4. Log results to file and display in console

    ```powershell
    # validate.ps1 - Matrix Shader Clean-Room Validation
    # Run in Windows Sandbox to verify no runtime dependencies

    $ErrorActionPreference = 'Stop'
    $TestDir = "C:\MatrixShader"
    $ResultsFile = "C:\Test\results.txt"

    Write-Host "================================" -ForegroundColor Cyan
    Write-Host " Matrix Shader Validation" -ForegroundColor Cyan
    Write-Host "================================" -ForegroundColor Cyan
    Write-Host ""

    $exes = @(
        @{ Name = "wakeupneo.exe"; Args = "--help" },
        @{ Name = "redpill.exe"; Args = "--help" },
        @{ Name = "bluepill.exe"; Args = "--help" }
    )

    $results = @()
    $allPassed = $true

    foreach ($exe in $exes) {
        $path = Join-Path $TestDir $exe.Name

        # Check existence
        if (!(Test-Path $path)) {
            Write-Host "FAIL: $($exe.Name) not found" -ForegroundColor Red
            $results += "FAIL: $($exe.Name) - NOT FOUND"
            $allPassed = $false
            continue
        }

        # Measure startup time
        $sw = [System.Diagnostics.Stopwatch]::StartNew()
        try {
            $proc = Start-Process -FilePath $path -ArgumentList $exe.Args -NoNewWindow -PassThru -Wait
            $sw.Stop()
            $ms = $sw.ElapsedMilliseconds

            if ($proc.ExitCode -ne 0) {
                Write-Host "FAIL: $($exe.Name) exited with code $($proc.ExitCode)" -ForegroundColor Red
                $results += "FAIL: $($exe.Name) - Exit code $($proc.ExitCode)"
                $allPassed = $false
            }
            elseif ($ms -gt 500) {
                Write-Host "FAIL: $($exe.Name) startup ${ms}ms (>500ms)" -ForegroundColor Red
                $results += "FAIL: $($exe.Name) - ${ms}ms (exceeded 500ms)"
                $allPassed = $false
            }
            else {
                Write-Host "PASS: $($exe.Name) startup ${ms}ms" -ForegroundColor Green
                $results += "PASS: $($exe.Name) - ${ms}ms"
            }
        }
        catch {
            $sw.Stop()
            Write-Host "FAIL: $($exe.Name) threw exception: $_" -ForegroundColor Red
            $results += "FAIL: $($exe.Name) - Exception: $_"
            $allPassed = $false
        }
    }

    # Summary
    Write-Host ""
    Write-Host "================================" -ForegroundColor Cyan
    if ($allPassed) {
        Write-Host " ALL TESTS PASSED" -ForegroundColor Green
    }
    else {
        Write-Host " SOME TESTS FAILED" -ForegroundColor Red
    }
    Write-Host "================================" -ForegroundColor Cyan

    # Save results
    $results | Out-File -FilePath $ResultsFile -Encoding UTF8
    Write-Host ""
    Write-Host "Results saved to: $ResultsFile" -ForegroundColor Gray

    # Keep window open
    Write-Host ""
    Write-Host "Press any key to close sandbox..." -ForegroundColor Yellow
    $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")

    if ($allPassed) { exit 0 } else { exit 1 }
    ```
