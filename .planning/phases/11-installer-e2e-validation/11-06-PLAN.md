Run dotnet publish for all 6 projects to rebuild the publish directory with Phase 11 fixes.

Execute from project root (C:\Users\ehome\Documents\Matrix):

```powershell
$PublishDir = "installer\publish"
$SrcRoot = "MatrixShader\src"

# Clean previous stale builds
if (Test-Path $PublishDir) { Remove-Item -Recurse -Force $PublishDir }
New-Item -ItemType Directory -Force -Path $PublishDir | Out-Null

# Publish all 6 projects
$projects = @(
    "MatrixShader.Cli\Bluepill",
    "MatrixShader.Cli\Redpill",
    "MatrixShader.Cli\WakeupNeo",
    "MatrixShader.Cli\MatrixLite",
    "MatrixShader.Hotkeys",
    "MatrixShader.Monitor"
)

foreach ($project in $projects) {
    $csproj = Join-Path $SrcRoot "$project\*.csproj"
    dotnet publish $csproj -c Release -o $PublishDir --self-contained true -r win-x64
    if ($LASTEXITCODE -ne 0) { throw "Failed to publish $project" }
}
```
